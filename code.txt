# preprocess_lyrics.py
import pandas as pd
import re
import string

# Fungsi preprocessing final
def preprocess_lyrics_final(text):
    if not isinstance(text, str):
        return ""

    # 1. Ambil dari bagian [Intro], [Verse], dll.
    match = re.search(r"\[(intro|verse|chorus|bridge|outro)[^\]]*\]", text, re.IGNORECASE)
    if match:
        text = text[match.start():]
    else:
        return ""

    # 2. Potong jika ada "Read More" di akhir
    text = re.split(r"read\s*more", text, flags=re.IGNORECASE)[0]

    # 3. Hapus tag seperti [Chorus], [Verse ...]
    text = re.sub(r"\[[^\]]*\]", "", text)

    # 4. Ubah ke huruf kecil
    text = text.lower()

    # 5. Hapus angka
    text = re.sub(r"\d+", "", text)

    # 6. Hapus tanda baca
    text = text.translate(str.maketrans("", "", string.punctuation))

    # 7. Hapus karakter non-ASCII (emoji, simbol khusus)
    text = text.encode('ascii', 'ignore').decode()

    # 8. Hapus spasi berlebih
    text = re.sub(r"\s+", " ", text).strip()

    return text

# Daftar pola regex untuk membersihkan judul lagu
title_patterns = [
    r' - Radio Edit', 
    r' - Live in.*', 
    r' - .*Remix', 
    r'\(.*?remix.*?\)', 
    r'$.*?$', 
    r'-\s*Bonus.*', 
    r'feat\..*', 
    r'ft\..*', 
    r'[^\w\s]', 
]

# Fungsi pembersihan judul lagu
def clean_title(title):
    for pattern in title_patterns:
        title = re.sub(pattern, '', title, flags=re.IGNORECASE)
    return title.strip().lower()

import pandas as pd
import re

# Daftar pola regex untuk membersihkan judul lagu
patterns = [
    r' \(-.*?\)',  # Hapus bagian "- Remix", "- Live", dll.
    r' \(.*?\)',   # Hapus bagian "(Remix)", "(Live)", dll.
    r' feat\..*',  # Hapus bagian "feat. Artis Lain"
    r' - .*',      # Hapus bagian "- Remix", "- PartyNextDoor", dll.
]

# Fungsi untuk membersihkan judul lagu
def clean_title(title):
    for pattern in patterns:
        title = re.sub(pattern, '', title, flags=re.IGNORECASE)
    return title.strip().lower()

# Baca dataset Excel
df = pd.read_excel(file_name)

# Filter baris dengan lirik tidak valid
df_cleaned = df[
    (~df['lirik'].str.contains("Lirik tidak ditemukan", case=False, na=True)) &
    (~df['lirik'].str.contains("error", case=False, na=True))
].copy()  # Gunakan .copy() untuk hindari warning

# Tambahkan judul lagu bersih sebagai field sementara (di memory saja)
df_cleaned['cleaned_title'] = df_cleaned['judul_lagu'].apply(lambda x: clean_title(x))

# Hapus duplikat berdasarkan judul_lagu yang telah dibersihkan
df_filtered = df_cleaned.drop_duplicates(subset=['cleaned_title'], keep='first')

import pandas as pd
from nltk.sentiment import SentimentIntensityAnalyzer
import nltk
import os

# nltk.download('vader_lexicon')

# Baca file Excel
file_name = "spotify_lirik_bersih.xlsx"
df = pd.read_excel(file_name)

# Filter baris dengan lirik tidak valid
df = df[df['lirik'].notna() & (df['lirik'] != '') & (df['lirik'] != 'Lirik tidak ditemukan')]

# Inisialisasi VADER
sia = SentimentIntensityAnalyzer()

# Fungsi untuk mendapatkan compound score dan label sentimen
def get_sentiment(text):
    score = sia.polarity_scores(text)['compound']
    if score > 0.05:
        return {'compound': score, 'sentimen': 'positif'}
    elif score < -0.05:
        return {'compound': score, 'sentimen': 'negatif'}
    else:
        return {'compound': score, 'sentimen': 'netral'}

# Terapkan fungsi ke setiap baris
results = df['lirik'].apply(lambda x: get_sentiment(x))

# Tambahkan hasil ke DataFrame
df['compound_score'] = results.apply(lambda x: x['compound'])
df['sentimen'] = results.apply(lambda x: x['sentimen'])

# Simpan ke file baru
output_file = "spotify_pop_music_with_sentiment.xlsx"
df.to_excel(output_file, index=False)

print(f"✅ Analisis selesai! File tersimpan sebagai: {output_file}")
